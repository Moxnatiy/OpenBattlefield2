function(obf2_add_test name)
  add_executable(${name} ${name}.cpp)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(${name} PRIVATE obf2_flags ${ARGN})
  add_test(NAME ${name} COMMAND ${name})
endfunction()

obf2_add_test(test_path obf2_core)
obf2_add_test(test_con obf2_con)
obf2_add_test(test_mesh obf2_mesh)
obf2_add_test(test_texture obf2_texture)
obf2_add_test(test_object_template obf2_game)
obf2_add_test(test_scene obf2_game)
obf2_add_test(test_level obf2_level)
obf2_add_test(test_engine obf2_engine)
